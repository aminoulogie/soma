  if (recovery >= 80) return "rgba(245, 158, 11, 0.7)"; // Amber
  if (recovery >= 50) return "rgba(249, 115, 22, 0.8)"; // Orange
  return "rgba(239, 68, 68, 0.85)";                      // Red
}

function renderDetails(key) {
  const model = muscleRegistry[key];
  if (!model) return;
  const color = getHeatColor(model.recovery);
  cardBox.innerHTML = `
    <div class="bm-card-subtitle">${model.region}</div>
    <div class="bm-card-title">${model.name}</div>
    <div class="bm-meta-row">
      <span>Strain Status:</span>
      <span class="bm-meta-val" style="color:${color}">${model.tier} (${model.recovery}%)</span>
    </div>
    <div class="bm-meta-row">
      <span>Recovery Window:</span>
      <span class="bm-meta-val">${model.defaultHours} hrs</span>
    </div>
    <div class="bm-desc-box">${model.desc}</div>
  `;
}

function renderMap() {
  viewport.empty();

  // Simplified vector silhouette container based on active view
  let pathsHtml = "";
  let firstKey = null;

  for (const [key, item] of Object.entries(muscleRegistry)) {
    if (item.view === currentView) {
      if (!firstKey) firstKey = key;
      const isSelected = key === selectedKey ? "selected" : "";
      const color = getHeatColor(item.recovery);
      pathsHtml += `<path class="bm-muscle-path ${isSelected}" data-key="${key}" d="${item.d}" fill="${color}" />`;
    }
  }

  if (!muscleRegistry[selectedKey] || muscleRegistry[selectedKey].view !== currentView) {
    selectedKey = firstKey || "";
  }

  // Base vector outline placeholder for body silhouette
  const silhouette = currentView === "front" 
    ? `<path class="bm-base-body" d="M 50,12 C 58,12 64,20 64,30 C 64,40 58,48 50,48 C 42,48 36,40 36,30 C 36,20 42,12 50,12 Z M 22,42 L 32,45 L 28,80 L 15,75 Z M 78,42 L 68,45 L 72,80 L 85,75 Z M 35,48 L 65,48 L 62,135 L 58,260 L 52,260 L 48,170 L 44,260 L 38,260 Z" />`
    : `<path class="bm-base-body" d="M 50,12 C 58,12 64,20 64,30 C 64,40 58,48 50,48 C 42,48 36,40 36,30 C 36,20 42,12 50,12 Z M 35,48 L 65,48 L 62,135 L 58,260 L 52,260 L 48,170 L 44,260 L 38,260 Z" />`;

  viewport.innerHTML = `
    <svg class="bm-vector-svg" viewBox="0 0 100 280" xmlns="http://www.w3.org/2000/svg">
      ${silhouette}
      ${pathsHtml}
    </svg>
  `;

  const paths = viewport.querySelectorAll(".bm-muscle-path");
  paths.forEach((p) => {
    const key = p.getAttribute("data-key");
    p.addEventListener("mouseenter", () => renderDetails(key));
    p.addEventListener("mouseleave", () => renderDetails(selectedKey));
    p.addEventListener("click", () => {
      selectedKey = key;
      paths.forEach(el => el.classList.remove("selected"));
      p.classList.add("selected");
      renderDetails(key);
    });
  });

  if (selectedKey && muscleRegistry[selectedKey]) {
    renderDetails(selectedKey);
  }
}

btnFront.onclick = () => {
  currentView = "front";
  selectedKey = "chest";
  btnFront.classList.add("active");
  btnBack.classList.remove("active");
  renderMap();
};

btnBack.onclick = () => {
  currentView = "back";
  selectedKey = "traps";
  btnBack.classList.add("active");
  btnFront.classList.remove("active");
  renderMap();
};

renderMap();
