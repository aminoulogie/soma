// 10. GEOMETRY HELPERS
function unionBBox(elements) {
    let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
    elements.forEach(el => {
        const b = el.getBBox();
        minX = Math.min(minX, b.x); minY = Math.min(minY, b.y);
        maxX = Math.max(maxX, b.x + b.width); maxY = Math.max(maxY, b.y + b.height);
    });
    return { x: minX, y: minY, width: maxX - minX, height: maxY - minY };
}
// Bands are still sized/stacked by the same cumulative fractions as before
// (so overall proportions are unchanged), but each band is now enlarged and
// tagged with a rotation pivoted on its own center (cx, cy). The clipPath
// (the muscle's real outline) trims away the overhang, so the visible cut
// between adjacent bands comes out as an angled line along the fiber
// direction instead of a flat horizontal/vertical box edge.
function computeBands(bandDefs, bbox, axis) {
    let cum = 0;
    const overflow = axis === "vertical" ? bbox.width * 0.9 : bbox.height * 0.9;
    return bandDefs.map((b, idx) => {
        const start = cum, end = cum + b.frac;
        cum = end;
        const tilt = b.angle * 0.35; // damped — a subtle organic tilt, not the full fiber-pattern angle
        if (axis === "vertical") {
            const y = bbox.y + start * bbox.height;
            const h = (end - start) * bbox.height;
            return {
                idx, name: b.name, tilt,
                x: bbox.x - overflow / 2, y, width: bbox.width + overflow, height: h,
                cx: bbox.x + bbox.width / 2, cy: y + h / 2,
            };
        }
        const x = bbox.x + start * bbox.width;
        const w = (end - start) * bbox.width;
        return {
            idx, name: b.name, tilt,
            x, y: bbox.y - overflow / 2, width: w, height: bbox.height + overflow,
            cx: x + w / 2, cy: bbox.y + bbox.height / 2,
        };
    });
}

    let pathsHtml = "";
    const outline = view === "front" ? FRONT_OUTLINE : BACK_OUTLINE;
    const vb = view === "front" ? FRONT_VIEWBOX : BACK_VIEWBOX;
    const staticParts = STATIC_PARTS[view];
    const defsHtml = buildDefs(view);

    for (const [partKey, partData] of Object.entries(staticParts)) {
        pathsHtml += `<path class="bm3-static-part" d="${partData.d}" fill="${partData.color}" />`;
    }

    for (const [key, item] of Object.entries(muscleRegistry)) {
        if (item.view !== view) continue;
        const gid = `grad-${INSTANCE_ID}-${view}-${key}`;
        const tier = getTier(item.recovery);
        const isSelected = key === selectedKey;
        const filtered = activePresetKeys && !activePresetKeys.has(key);
        // A muscle the user explicitly locked in, even though it's outside
        // the current preset filter, always shows fully selected in blue —
        // never falls back to the flat inactive grey.
        const showAsSelectedInactive = isSelected && filtered;
        const isInactiveHoverable = filtered && !isSelected;
        const fill = showAsSelectedInactive ? `url(#blueSelectGrad-${INSTANCE_ID})` : (filtered ? INACTIVE_FILL : `url(#${gid})`);
        const stroke = showAsSelectedInactive ? SELECT_BLUE : (filtered ? INACTIVE_STROKE : tier.dark);
        const pathClass = `bm3-muscle-path${isSelected ? (showAsSelectedInactive ? " selected selected-inactive" : " selected") : ""}${isInactiveHoverable ? " inactive" : ""}`;

        pathsHtml += `<g class="bm3-muscle-group${isSelected ? " selected" : ""}">`;
        item.paths.forEach(p => {
            pathsHtml += `<path class="${pathClass}" data-key="${key}" d="${p}" fill="${fill}" stroke="${stroke}" stroke-width="0.6" style="color:${tier.base}" />`;
        });

        if ((!filtered || showAsSelectedInactive) && !ZONE_DEFS[key]) {
            const pid = `fiber-${INSTANCE_ID}-${view}-${key}`;
            item.paths.forEach(p => {
                pathsHtml += `<path class="bm3-fiber-overlay${isSelected ? " selected" : ""}" data-key="${key}" d="${p}" fill="url(#${pid})" />`;
            });
        }
        pathsHtml += `</g>`;
    }

    viewport.innerHTML = `
        <svg class="bm3-vector-svg" viewBox="${vb}" xmlns="http://www.w3.org/2000/svg">
            <defs>${defsHtml}</defs>
            <path class="bm3-base-body" d="${outline}" />
            ${pathsHtml}
        </svg>
    `;

    // Reading-mode fix: force a reflow so gradient/pattern refs bind on first paint.
    void viewport.offsetHeight;
    const svgEl = viewport.querySelector(".bm3-vector-svg");
    if (svgEl) {
        requestAnimationFrame(() => {
            svgEl.style.display = "none";
            void svgEl.offsetHeight;
            svgEl.style.display = "";
        });
    }
    if (!svgEl) return;

    // --- Phase B: build real clipped sub-zones now that elements exist in the DOM ---
    for (const [key, item] of Object.entries(muscleRegistry)) {
        if (item.view !== view) continue;
        const zoneDef = ZONE_DEFS[key];
        if (!zoneDef) continue;
        const filtered = activePresetKeys && !activePresetKeys.has(key);
        if (filtered) continue;

        const gid = `grad-${INSTANCE_ID}-${view}-${key}`;
        const clipId = `clip-${INSTANCE_ID}-${view}-${key}`;
        const tier = getTier(item.recovery);
        const baseEls = Array.from(svgEl.querySelectorAll(`.bm3-muscle-path[data-key="${key}"]`));
        if (!baseEls.length) continue;
        const bbox = unionBBox(baseEls);
        const bands = computeBands(zoneDef.bands, bbox, zoneDef.axis);

        let zonesHtml = `<g clip-path="url(#${clipId})">`;
        bands.forEach(band => {
            const pid = `zfiber-${INSTANCE_ID}-${view}-${key}-${band.idx}`;
            const isSel = selectedZone && selectedZone.key === key && selectedZone.idx === band.idx;
            const tiltVar = `--tilt:${band.tilt}deg;`;
            zonesHtml += `<rect class="bm3-zone-texture${isSel ? " selected" : ""}" style="${tiltVar}" x="${band.x}" y="${band.y}" width="${band.width}" height="${band.height}" fill="url(#${pid})" />`;
            zonesHtml += `<rect class="bm3-zone-hit${isSel ? " selected" : ""}" data-parent="${key}" data-zone="${band.idx}" data-name="${band.name}" data-cx="${band.cx}" data-cy="${band.cy}"
                x="${band.x}" y="${band.y}" width="${band.width}" height="${band.height}" fill="url(#${gid})" style="${tiltVar}color:${tier.base}" />`;
        });
        zonesHtml += `</g>`;
        // Whole-muscle contour outline when one of its zones is selected —
        // drawn on the real (unclipped) outline, so there's no hard rectangle
        // edge between adjacent bands. Colored to match this muscle's own
        // fatigue tier (dimmed), not a fixed blue, so a red/fatigued muscle
        // still reads as red when a sub-zone is picked.
        if (selectedZone && selectedZone.key === key) {
            const oc = tier.base;
            zonesHtml += item.paths.map(p => `<path class="bm3-zone-outline" d="${p}" style="stroke:${oc}; opacity:0.55; filter:drop-shadow(0 0 6px ${oc}99);" />`).join("");
        }
        svgEl.insertAdjacentHTML("beforeend", zonesHtml);
    }

    // --- Attach listeners: whole-muscle paths ---
    svgEl.querySelectorAll(".bm3-muscle-path").forEach((p) => {
        const key = p.getAttribute("data-key");
        p.addEventListener("mouseenter", () => renderDetails(key));
        p.addEventListener("mouseleave", () => {
            if (selectedZone) renderDetails(selectedZone.key, selectedZone.name);
            else if (selectedKey) renderDetails(selectedKey);
        });
        p.addEventListener("click", () => {
            selectedKey = key;
            selectedZone = null;
            const b = p.getBBox();
            selectedPoint = { x: b.x + b.width / 2, y: b.y + b.height / 2 };
            renderBody(currentView);
            renderDetails(key);
        });
    });

    // --- Attach listeners: zone hit rects (take priority, sit on top) ---
    svgEl.querySelectorAll(".bm3-zone-hit").forEach((r) => {
        const key = r.getAttribute("data-parent");
        const idx = parseInt(r.getAttribute("data-zone"), 10);
        const name = r.getAttribute("data-name");
        r.addEventListener("mouseenter", () => renderDetails(key, name));
        r.addEventListener("mouseleave", () => {
            if (selectedZone) renderDetails(selectedZone.key, selectedZone.name);
            else if (selectedKey) renderDetails(selectedKey);
        });
        r.addEventListener("click", (e) => {
            e.stopPropagation();
            selectedZone = { key, idx, name };
            selectedKey = null;
            selectedPoint = { x: parseFloat(r.getAttribute("data-cx")), y: parseFloat(r.getAttribute("data-cy")) };
            renderBody(currentView);
            renderDetails(key, name);
        });
    });

    // Re-draw the annotation after a rebuild if something is still selected —
    // uses the exact stored click point, not a recomputed bounding box.
    if (selectedZone && selectedPoint) {
        showAnnotation(svgEl, view, selectedPoint, selectedZone.name);
    } else if (selectedKey && selectedPoint) {
        showAnnotation(svgEl, view, selectedPoint, muscleRegistry[selectedKey].name);
    }
}

function switchView(view) {
    currentView = view;
    selectedKey = null;
    selectedZone = null;
    selectedPoint = null;
    frontTab.classList.toggle("active", view === "front");
    backTab.classList.toggle("active", view === "back");
    renderBody(view);
    const firstKey = Object.keys(muscleRegistry).find(k => muscleRegistry[k].view === view);
    if (firstKey) renderDetails(firstKey);
}

frontTab.addEventListener("click", () => switchView("front"));
backTab.addEventListener("click", () => switchView("back"));

    if (selectedZone) renderDetails(selectedZone.key, selectedZone.name);
    else if (selectedKey) renderDetails(selectedKey);
}
