// 7. RENDER BODY WITH PER-MUSCLE RADIAL GRADIENTS (fakes 3D bulge/highlight)
let gradCounter = 0;
function buildGradientDefs(view) {
    let defs = "";
    for (const [key, item] of Object.entries(muscleRegistry)) {
        if (item.view !== view) continue;
        const tier = getTier(item.recovery);
        const gid = `grad-${view}-${key}`;
        // Radial gradient: highlight upper-left, base mid, dark rim edge -> convex bulge illusion
        defs += `
            <radialGradient id="${gid}" cx="35%" cy="30%" r="75%">
                <stop offset="0%" stop-color="${tier.light}" stop-opacity="0.95" />
                <stop offset="45%" stop-color="${tier.base}" stop-opacity="0.95" />
                <stop offset="100%" stop-color="${tier.dark}" stop-opacity="0.98" />
            </radialGradient>
        `;
    }
    return defs;
}

function renderBody(viewport, view) {
    viewport.empty();
    let pathsHtml = "";
    const outline = view === "front" ? FRONT_OUTLINE : BACK_OUTLINE;
    const vb = view === "front" ? FRONT_VIEWBOX : BACK_VIEWBOX;
    const staticParts = STATIC_PARTS[view];
    const defsHtml = buildGradientDefs(view);

    for (const [partKey, partData] of Object.entries(staticParts)) {
        pathsHtml += `<path class="bm2-static-part" d="${partData.d}" fill="${partData.color}" />`;
    }

    for (const [key, item] of Object.entries(muscleRegistry)) {
        if (item.view === view) {
            const gid = `grad-${view}-${key}`;
            const tier = getTier(item.recovery);
            item.paths.forEach(p => {
                pathsHtml += `<path class="bm2-muscle-path" data-key="${key}" d="${p}" fill="url(#${gid})" stroke="${tier.dark}" stroke-width="0.6" style="color:${tier.base}" />`;
            });
        }
    }

    viewport.innerHTML = `
        <svg class="bm2-vector-svg" viewBox="${vb}" xmlns="http://www.w3.org/2000/svg">
            <defs>${defsHtml}</defs>
            <path class="bm2-base-body" d="${outline}" />
            ${pathsHtml}
        </svg>
    `;

    const paths = viewport.querySelectorAll(".bm2-muscle-path");
    paths.forEach((p) => {
        const key = p.getAttribute("data-key");
        p.addEventListener("mouseenter", () => renderDetails(key));
        p.addEventListener("click", () => {
            document.querySelectorAll(".bm2-muscle-path.selected").forEach(el => el.classList.remove("selected"));
            p.classList.add("selected");
            renderDetails(key);
        });
    });
}
