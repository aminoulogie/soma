// 8. DEFS: gradients (per muscle) + fiber patterns (per muscle, or per zone-band) + clipPaths (zone muscles)
function buildDefs(view) {
    let defs = "";
    for (const [key, item] of Object.entries(muscleRegistry)) {
        if (item.view !== view) continue;
        const tier = getTier(item.recovery);
        const gid = `grad-${INSTANCE_ID}-${view}-${key}`;

        defs += `
            <radialGradient id="${gid}" cx="32%" cy="26%" r="80%">
                <stop offset="0%"  stop-color="${tier.light}" stop-opacity="1" />
                <stop offset="30%" stop-color="${tier.base}"  stop-opacity="1" />
                <stop offset="62%" stop-color="${tier.base}"  stop-opacity="0.96" />
                <stop offset="85%" stop-color="${tier.dark}"  stop-opacity="0.97" />
                <stop offset="100%" stop-color="${tier.dark}" stop-opacity="1" />
            </radialGradient>
        `;

        const zoneDef = ZONE_DEFS[key];
        if (zoneDef) {
            // clipPath uses the muscle's real outline paths directly - no bbox needed
            const clipId = `clip-${INSTANCE_ID}-${view}-${key}`;
            defs += `<clipPath id="${clipId}">${item.paths.map(p => `<path d="${p}" />`).join("")}</clipPath>`;
            zoneDef.bands.forEach((band, idx) => {
                const pid = `zfiber-${INSTANCE_ID}-${view}-${key}-${idx}`;
                defs += `
                    <pattern id="${pid}" width="6" height="6" patternTransform="rotate(${band.angle})" patternUnits="userSpaceOnUse">
                        <rect width="6" height="6" fill="transparent" />
                        <line x1="0" y1="0" x2="0" y2="6" stroke="#000000" stroke-width="0.7" stroke-opacity="0.55" />
                        <line x1="3" y1="0" x2="3" y2="6" stroke="#ffffff" stroke-width="0.35" stroke-opacity="0.22" />
                    </pattern>
                `;
            });
        } else {
            const pid = `fiber-${INSTANCE_ID}-${view}-${key}`;
            defs += `
                <pattern id="${pid}" width="5" height="5" patternTransform="rotate(58)" patternUnits="userSpaceOnUse">
                    <rect width="5" height="5" fill="transparent" />
                    <line x1="0" y1="0" x2="0" y2="5" stroke="#000000" stroke-width="0.55" stroke-opacity="0.5" />
                    <line x1="2.5" y1="0" x2="2.5" y2="5" stroke="#ffffff" stroke-width="0.3" stroke-opacity="0.18" />
                </pattern>
            `;
        }
    }
    return defs;
}

// 10. GEOMETRY HELPERS
function unionBBox(elements) {
    let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
    elements.forEach(el => {
        const b = el.getBBox();
        minX = Math.min(minX, b.x); minY = Math.min(minY, b.y);
        maxX = Math.max(maxX, b.x + b.width); maxY = Math.max(maxY, b.y + b.height);
    });
    return { x: minX, y: minY, width: maxX - minX, height: maxY - minY };
}
function computeBands(bandDefs, bbox, axis) {
    let cum = 0;
    return bandDefs.map((b, idx) => {
        const start = cum, end = cum + b.frac;
        cum = end;
        return axis === "vertical"
            ? { idx, name: b.name, x: bbox.x, y: bbox.y + start * bbox.height, width: bbox.width, height: (end - start) * bbox.height }
            : { idx, name: b.name, x: bbox.x + start * bbox.width, y: bbox.y, width: (end - start) * bbox.width, height: bbox.height };
    });
}

// 11. RENDER BODY (single active view, front or back)
function renderBody(view) {
    viewport.empty();
    let pathsHtml = "";
    const outline = view === "front" ? FRONT_OUTLINE : BACK_OUTLINE;
    const vb = view === "front" ? FRONT_VIEWBOX : BACK_VIEWBOX;
    const staticParts = STATIC_PARTS[view];
    const defsHtml = buildDefs(view);

    for (const [partKey, partData] of Object.entries(staticParts)) {
        pathsHtml += `<path class="bm3-static-part" d="${partData.d}" fill="${partData.color}" />`;
    }

    for (const [key, item] of Object.entries(muscleRegistry)) {
        if (item.view !== view) continue;
        const gid = `grad-${INSTANCE_ID}-${view}-${key}`;
        const tier = getTier(item.recovery);
        const isSelected = key === selectedKey;
        const filtered = activePresetKeys && !activePresetKeys.has(key);
        const fill = filtered ? INACTIVE_FILL : `url(#${gid})`;
        const stroke = filtered ? INACTIVE_STROKE : tier.dark;

        item.paths.forEach(p => {
            pathsHtml += `<path class="bm3-muscle-path${isSelected ? " selected" : ""}" data-key="${key}" d="${p}" fill="${fill}" stroke="${stroke}" stroke-width="0.6" style="color:${tier.base}" />`;
        });

        if (!filtered && !ZONE_DEFS[key]) {
            const pid = `fiber-${INSTANCE_ID}-${view}-${key}`;
            item.paths.forEach(p => {
                pathsHtml += `<path class="bm3-fiber-overlay${isSelected ? " selected" : ""}" data-key="${key}" d="${p}" fill="url(#${pid})" />`;
            });
        }
    }

    viewport.innerHTML = `
        <svg class="bm3-vector-svg" viewBox="${vb}" xmlns="http://www.w3.org/2000/svg">
            <defs>${defsHtml}</defs>
            <path class="bm3-base-body" d="${outline}" />
            ${pathsHtml}
        </svg>
    `;

    // Reading-mode fix: force a reflow so gradient/pattern refs bind on first paint.
    void viewport.offsetHeight;
    const svgEl = viewport.querySelector(".bm3-vector-svg");
    if (svgEl) {
        requestAnimationFrame(() => {
            svgEl.style.display = "none";
            void svgEl.offsetHeight;
            svgEl.style.display = "";
        });
    }
    if (!svgEl) return;

    // --- Phase B: build real clipped sub-zones now that elements exist in the DOM ---
    for (const [key, item] of Object.entries(muscleRegistry)) {
        if (item.view !== view) continue;
        const zoneDef = ZONE_DEFS[key];
        if (!zoneDef) continue;
        const filtered = activePresetKeys && !activePresetKeys.has(key);
        if (filtered) continue;

        const gid = `grad-${INSTANCE_ID}-${view}-${key}`;
        const clipId = `clip-${INSTANCE_ID}-${view}-${key}`;
        const tier = getTier(item.recovery);
        const baseEls = Array.from(svgEl.querySelectorAll(`.bm3-muscle-path[data-key="${key}"]`));
        if (!baseEls.length) continue;
        const bbox = unionBBox(baseEls);
        const bands = computeBands(zoneDef.bands, bbox, zoneDef.axis);

        let zonesHtml = `<g clip-path="url(#${clipId})">`;
        bands.forEach(band => {
            const pid = `zfiber-${INSTANCE_ID}-${view}-${key}-${band.idx}`;
            const isSel = selectedZone && selectedZone.key === key && selectedZone.idx === band.idx;
            zonesHtml += `<rect class="bm3-zone-texture" x="${band.x}" y="${band.y}" width="${band.width}" height="${band.height}" fill="url(#${pid})" />`;
            zonesHtml += `<rect class="bm3-zone-hit${isSel ? " selected" : ""}" data-parent="${key}" data-zone="${band.idx}" data-name="${band.name}"
                x="${band.x}" y="${band.y}" width="${band.width}" height="${band.height}" fill="url(#${gid})" style="color:${tier.base}" />`;
        });
        zonesHtml += `</g>`;
        svgEl.insertAdjacentHTML("beforeend", zonesHtml);
    }

    // --- Attach listeners: whole-muscle paths ---
    svgEl.querySelectorAll(".bm3-muscle-path").forEach((p) => {
        const key = p.getAttribute("data-key");
        p.addEventListener("mouseenter", () => renderDetails(key));
        p.addEventListener("mouseleave", () => {
            if (selectedZone) renderDetails(selectedZone.key, selectedZone.name);
            else if (selectedKey) renderDetails(selectedKey);
        });
        p.addEventListener("click", () => {
            selectedKey = key;
            selectedZone = null;
            renderBody(currentView);
            renderDetails(key);
            const el = svgEl.querySelector(`.bm3-muscle-path[data-key="${key}"]`);
            if (el) showAnnotation(svgEl, currentView, el.getBBox(), muscleRegistry[key].name);
        });
    });

    // --- Attach listeners: zone hit rects (take priority, sit on top) ---
    svgEl.querySelectorAll(".bm3-zone-hit").forEach((r) => {
        const key = r.getAttribute("data-parent");
        const idx = parseInt(r.getAttribute("data-zone"), 10);
        const name = r.getAttribute("data-name");
        r.addEventListener("mouseenter", () => renderDetails(key, name));
        r.addEventListener("mouseleave", () => {
            if (selectedZone) renderDetails(selectedZone.key, selectedZone.name);
            else if (selectedKey) renderDetails(selectedKey);
        });
        r.addEventListener("click", (e) => {
            e.stopPropagation();
            selectedZone = { key, idx, name };
            selectedKey = null;
            renderBody(currentView);
            renderDetails(key, name);
            const el = svgEl.querySelector(`.bm3-zone-hit[data-parent="${key}"][data-zone="${idx}"]`);
            if (el) showAnnotation(svgEl, currentView, el.getBBox(), name);
        });
    });

    // Re-draw annotation line after a rebuild if something is still selected
    if (selectedZone) {
        const el = svgEl.querySelector(`.bm3-zone-hit[data-parent="${selectedZone.key}"][data-zone="${selectedZone.idx}"]`);
        if (el) showAnnotation(svgEl, view, el.getBBox(), selectedZone.name);
    } else if (selectedKey) {
        const el = svgEl.querySelector(`.bm3-muscle-path[data-key="${selectedKey}"]`);
        if (el) showAnnotation(svgEl, view, el.getBBox(), muscleRegistry[selectedKey].name);
    }
}

function switchView(view) {
    currentView = view;
    selectedKey = null;
    selectedZone = null;
    frontTab.classList.toggle("active", view === "front");
    backTab.classList.toggle("active", view === "back");
    renderBody(view);
    const firstKey = Object.keys(muscleRegistry).find(k => muscleRegistry[k].view === view);
    if (firstKey) renderDetails(firstKey);
}

backTab.addEventListener("click", () => switchView("back"));
